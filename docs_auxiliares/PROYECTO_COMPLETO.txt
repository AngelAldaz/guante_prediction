🚀 CÓMO USAR EL SISTEMA
═══════════════════════════════════════════════════════════════════════════════

1️⃣  INSTALACIÓN:
    
    pip install pandas numpy scikit-learn

2️⃣  EJECUTAR TODO EL PIPELINE:
    
    python generar_datos_lsm.py
    python comparar_modelos.py
    python demo_predictor.py

3️⃣  USO SIMPLE EN CÓDIGO:
    
    from predictor_lsm import PredictorLSM
    
    predictor = PredictorLSM()
    letra = predictor.predecir(100, 0, 0, 0, 0, False)
    print(letra)  # Output: 'a'

4️⃣  DESDE FORMATO CSV (Arduino/Sensores):
    
    linea = "999,0,0,0,999,false"  # De sensores
    letra = predictor.predecir_desde_csv_linea(linea)
    print(letra)  # Output: 'y'

5️⃣  CON PROBABILIDADES:
    
    letra, probs = predictor.predecir_con_probabilidades(
        dedo1=0, dedo2=999, dedo3=999, dedo4=999, dedo5=999,
        movimiento=False
    )
    print(f"Letra: {letra} (confianza: {list(probs.values())[0]*100:.1f}%)")

🔌 INTEGRACIÓN CON HARDWARE
═══════════════════════════════════════════════════════════════════════════════

🐍 PYTHON (Receptor):

    from predictor_lsm import PredictorLSM
    import serial
    
    predictor = PredictorLSM()
    ser = serial.Serial('COM3', 9600, timeout=1)
    
    print("🔴 Escuchando señales...")
    
    while True:
        if ser.in_waiting > 0:
            linea = ser.readline().decode('utf-8').strip()
            letra = predictor.predecir_desde_csv_linea(linea)
            print(f"→ Letra detectada: {letra.upper()}")
            ser.write(f"{letra}\n".encode())  # Enviar respuesta

🔧 ARDUINO (Transmisor):

    void setup() {
        Serial.begin(9600);
        // Configurar pines de sensores
    }
    
    void loop() {
        int dedo1 = map(analogRead(A0), 0, 1023, 0, 999);
        int dedo2 = map(analogRead(A1), 0, 1023, 0, 999);
        int dedo3 = map(analogRead(A2), 0, 1023, 0, 999);
        int dedo4 = map(analogRead(A3), 0, 1023, 0, 999);
        int dedo5 = map(analogRead(A4), 0, 1023, 0, 999);
        bool movimiento = detectarMovimiento();
        
        // Formato CSV: dedo1,dedo2,dedo3,dedo4,dedo5,movimiento
        Serial.print(dedo1); Serial.print(",");
        Serial.print(dedo2); Serial.print(",");
        Serial.print(dedo3); Serial.print(",");
        Serial.print(dedo4); Serial.print(",");
        Serial.print(dedo5); Serial.print(",");
        Serial.println(movimiento ? "true" : "false");
        
        delay(500);
    }
